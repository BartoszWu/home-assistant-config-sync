#include <tunables/global>

# Custom AppArmor profile for HA Config Sync — Import.
# Mount: homeassistant_config → /homeassistant (RW). Application allowlist is
# still authoritative; this profile is defense-in-depth on the filesystem.
#
# Profile name must match config.yaml slug. Supervisor loads import/apparmor.txt.
#
# Deny mode letters (apparmor.d(5)):
#   r read, w write, m executable mmap, l hard link, k lock, x execute (bare x
#   only valid with deny). Append `a` conflicts with `w`, so sensitive paths get
#   a second deny rule for `a`.

  #include <abstractions/python>

  /package/** ix,
  /command/** ix,

  /etc/cont-init.d/** rwix,
  /etc/cont-finish.d/** rwix,

  /dev/tty rw,
  /usr/lib/bashio/** ix,
  /tmp/** rwk,
  /data/** rwk,

  # App code and Python deps
  /app.py r,
  /dashboard_logic.py r,
  /managed_files.py r,
  /managed_files.yaml r,
  /git_source.py r,
  /deployment_provenance.py r,
  /security.py r,

  /static/** r,
  /usr/lib/python*/** mr,
  /usr/local/lib/python*/** mr,
  /usr/lib/libpython* mr,

  # Read-only GitHub deploy key material (addon_config → /review)
  /review/** r,
  /review/ssh/** r,

  /usr/bin/git ix,
  /usr/bin/ssh ix,
  /usr/bin/ssh-keyscan ix,
  /etc/ssl/certs/** r,
  /etc/passwd r,

  /homeassistant/packages/ rw,
  /homeassistant/packages/** rw,

  deny /homeassistant/.storage/ a,
  deny /homeassistant/.storage/** rwmlkx,
  deny /homeassistant/.storage/** a,

  deny /homeassistant/secrets.yaml a,
  deny /homeassistant/configuration.yaml rwmlkx,
  deny /homeassistant/configuration.yaml a,
